Quick start

Register and submit your first endpoint in under a minute.

# 1. Register as a provider
curl -X POST https://api.claw-net.org/v1/providers/register \
  -H "X-API-Key: cn-your-api-key" \
  -H "Content-Type: application/json" \
  -d '{
    "name": "Acme Data",
    "slug": "acme-data",
    "email": "team@acme.com"
  }'

# 2. Submit an endpoint (goes live immediately)
curl -X POST https://api.claw-net.org/v1/providers/PROVIDER_ID/endpoints/submit \
  -H "X-API-Key: cn-your-api-key" \
  -H "Content-Type: application/json" \
  -d '{
    "name": "Token Price",
    "description": "Get real-time token prices with 24h change and volume",
    "category": "defi",
    "baseUrl": "https://api.acme.com",
    "path": "/v1/token/price",
    "costPerCall": 0.001
  }'

# 3. Check your earnings
curl https://api.claw-net.org/v1/providers/PROVIDER_ID/analytics \
  -H "X-API-Key: cn-your-api-key"
